    function updateIndicator(element) {

      if (!element || window.getComputedStyle(navLinks).display === 'none') {

        indicator.style.opacity = '0';

        return;

      }



      const parentRect = navLinks.getBoundingClientRect();

      const rect = element.getBoundingClientRect();



      // 精确校准当前悬停或激活菜单项的相对偏移

      const leftOffset = rect.left - parentRect.left;

      const topOffset = rect.top - parentRect.top;



      indicator.style.width = `${rect.width}px`;

      indicator.style.height = `${rect.height}px`;

      indicator.style.transform = `translate3d(${leftOffset}px, ${topOffset}px, 0)`;

      indicator.style.opacity = '1';

    }

      item.addEventListener('mouseenter', () => {

        updateIndicator(item);

      });



      item.addEventListener('click', (e) => {
